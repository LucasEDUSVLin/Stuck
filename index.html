function parseHubReceivedTime(value){ const s=String(value||'').trim(); if(!s) return null; const patterns=[ /^(\d{1,2})[-/](\d{1,2})[-/](\d{4})\s+(\d{1,2}):(\d{2})(?::(\d{2}))?$/, /^(\d{4})[-/](\d{1,2})[-/](\d{1,2})\s+(\d{1,2}):(\d{2})(?::(\d{2}))?$/, /^(\d{1,2})[-/](\d{1,2})[-/](\d{4})$/, /^(\d{4})[-/](\d{1,2})[-/](\d{1,2})$/ ]; for(const pattern of patterns){ const m=s.match(pattern); if(!m) continue; let dd, mo, yy, hh=0, min=0, sec=0; if(pattern.source.includes('\\d{4}\\s+')){ if(m.length >= 6){ if(m[1].length === 4){ yy=Number(m[1]); mo=Number(m[2]); dd=Number(m[3]); hh=Number(m[4]); min=Number(m[5]); sec=Number(m[6] || '0'); } else { dd=Number(m[1]); mo=Number(m[2]); yy=Number(m[3]); hh=Number(m[4]); min=Number(m[5]); sec=Number(m[6] || '0'); } } } else { if(m.length >= 4){ if(m[3].length === 4){ dd=Number(m[1]); mo=Number(m[2]); yy=Number(m[3]); } else { yy=Number(m[1]); mo=Number(m[2]); dd=Number(m[3]); } } } if(mo<1 || mo>12 || dd<1 || hh>23 || min>59 || sec>59) continue; const date=new Date(yy, mo-1, dd, hh, min, sec); if(date.getFullYear()===yy && date.getMonth()===mo-1 && date.getDate()===dd){ return date; } } return null; } function calculateDaysInHub(value){ const received=parseHubReceivedTime(value); if(!received) return null; const start=new Date(received.getFullYear(), received.getMonth(), received.getDate()); const today=new Date(); const todayStart=new Date(today.getFullYear(), today.getMonth(), today.getDate()); const elapsed=todayStart.getTime()-start.getTime(); return elapsed<0 ? null : Math.floor(elapsed / 86400000); } function initializeIcons(){ if(window.lucide && typeof window.lucide.createIcons === 'function') { window.lucide.createIcons(); } } if(document.readyState === 'loading'){ document.addEventListener('DOMContentLoaded', initializeIcons); } else { initializeIcons(); }
